cmake_minimum_required(VERSION 3.16)
project(Studbook LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(studbook_core src/database.cpp src/console.cpp)
target_include_directories(studbook_core PUBLIC src)
if(MSVC)
    target_compile_options(studbook_core PRIVATE /W4)
else()
    target_compile_options(studbook_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(studbook src/main.cpp)
target_link_libraries(studbook PRIVATE studbook_core)
enable_testing()
add_executable(studbook_tests tests/database_tests.cpp)
target_link_libraries(studbook_tests PRIVATE studbook_core)
add_test(NAME database_tests COMMAND studbook_tests)
